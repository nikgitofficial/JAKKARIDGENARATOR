import { useCallback, useEffect, useRef, useState } from "react";
import { PHOTO_ASPECT } from "@/lib/cardGeometry";

export type Photo = { bytes: Uint8Array; url: string };

const EXPORT_W = 480;
const EXPORT_H = Math.round(EXPORT_W / PHOTO_ASPECT);
const EDIT_W = 220;
const EDIT_H = Math.round(EDIT_W / PHOTO_ASPECT);

type View = { zoom: number; ox: number; oy: number }; // offsets in fractions of the frame

function draw(canvas: HTMLCanvasElement, img: HTMLImageElement, v: View) {
  const ctx = canvas.getContext("2d")!;
  const cw = canvas.width;
  const ch = canvas.height;
  const base = Math.max(cw / img.naturalWidth, ch / img.naturalHeight);
  const s = base * v.zoom;
  const w = img.naturalWidth * s;
  const h = img.naturalHeight * s;
  // keep the image covering the frame
  const maxX = (w - cw) / 2;
  const maxY = (h - ch) / 2;
  const dx = (cw - w) / 2 + Math.max(-maxX, Math.min(maxX, v.ox * cw));
  const dy = (ch - h) / 2 + Math.max(-maxY, Math.min(maxY, v.oy * ch));
  ctx.fillStyle = "#fff";
  ctx.fillRect(0, 0, cw, ch);
  ctx.imageSmoothingQuality = "high";
  ctx.drawImage(img, dx, dy, w, h);
}

function clampView(img: HTMLImageElement, v: View): View {
  const base = Math.max(EDIT_W / img.naturalWidth, EDIT_H / img.naturalHeight);
  const s = base * v.zoom;
  const mx = (img.naturalWidth * s - EDIT_W) / 2 / EDIT_W;
  const my = (img.naturalHeight * s - EDIT_H) / 2 / EDIT_H;
  return { zoom: v.zoom, ox: Math.max(-mx, Math.min(mx, v.ox)), oy: Math.max(-my, Math.min(my, v.oy)) };
}

export default function PhotoField({ onChange }: { onChange: (p: Photo | null) => void }) {
  const [img, setImg] = useState<HTMLImageElement | null>(null);
  const [view, setView] = useState<View>({ zoom: 1, ox: 0, oy: 0 });
  const [error, setError] = useState("");
  const editRef = useRef<HTMLCanvasElement>(null);
  const drag = useRef<{ x: number; y: number; ox: number; oy: number } | null>(null);
  const lastUrl = useRef<string | null>(null);

  const emit = useCallback(
    (image: HTMLImageElement, v: View) => {
      const out = document.createElement("canvas");
      out.width = EXPORT_W;
      out.height = EXPORT_H;
      draw(out, image, v);
      out.toBlob(async (blob) => {
        if (!blob) return;
        if (lastUrl.current) URL.revokeObjectURL(lastUrl.current);
        const url = URL.createObjectURL(blob);
        lastUrl.current = url;
        onChange({ bytes: new Uint8Array(await blob.arrayBuffer()), url });
      }, "image/png");
    },
    [onChange],
  );

  // Redraw the editor and (debounced) re-export whenever the framing changes.
  useEffect(() => {
    if (!img || !editRef.current) return;
    draw(editRef.current, img, view);
    const t = setTimeout(() => emit(img, view), 120);
    return () => clearTimeout(t);
  }, [img, view, emit]);

  useEffect(
    () => () => {
      if (lastUrl.current) URL.revokeObjectURL(lastUrl.current);
    },
    [],
  );

  function load(file: File | undefined) {
    if (!file) return;
    setError("");
    const src = URL.createObjectURL(file);
    const image = new Image();
    image.onload = () => {
      URL.revokeObjectURL(src);
      setImg(image);
      setView({ zoom: 1, ox: 0, oy: 0 });
    };
    image.onerror = () => {
      URL.revokeObjectURL(src);
      setError("That image couldn't be read. Please use a JPG or PNG file.");
    };
    image.src = src;
  }

  function remove() {
    setImg(null);
    setError("");
    if (lastUrl.current) URL.revokeObjectURL(lastUrl.current);
    lastUrl.current = null;
    onChange(null);
  }

  return (
    <div className="photo-field">
      {img ? (

            width={EDIT_W}
            height={EDIT_H}
            className="photo-edit"

              e.currentTarget.setPointerCapture(e.pointerId);
              drag.current = { x: e.clientX, y: e.clientY, ox: view.ox, oy: view.oy };
            }}
            onPointerMove={(e) => {
              const d = drag.current;
              if (!d || !img) return;
              const r = e.currentTarget.getBoundingClientRect();
              setView((v) =>
                clampView(img, {
                  zoom: v.zoom,
                  ox: d.ox + (e.clientX - d.x) / r.width,
                  oy: d.oy + (e.clientY - d.y) / r.height,
                }),
              );
            }}
            onPointerUp={() => (drag.current = null)}
            onPointerCancel={() => (drag.current = null)}
          />
          <label className="zoom">
            <span>Zoom</span>
            <input
              type="range"
              min={1}
              max={3}
              step={0.01}
              value={view.zoom}
              onChange={(e) => setView((v) => clampView(img, { ...v, zoom: Number(e.target.value) }))}
            />
          </label>
          <p className="hint">Drag the photo to reposition it.</p>
          <button type="button" className="btn-link" onClick={remove}>
            Remove photo
          </button>
        </>
      ) : (
        <label className="dropzone">
          <input type="file" accept="image/png,image/jpeg,image/webp" onChange={(e) => load(e.target.files?.[0])} />
          <span>Choose a photo</span>
          <small>JPG or PNG · cropped to the ID frame</small>
        </label>
      )}
      {error && (
        <p className="error" role="alert">
          {error}
        </p>
      )}
    </div>
  );

/** Apply a 0/1 mask like "XX-XXXXXXX-X" to a run of digits. */
function mask(s: string, pattern: string) {
  const d = digits(s);
  let out = "";
  let i = 0;
  for (const ch of pattern) {
    if (i >= d.length) break;
    if (ch === "X") out += d[i++];
    else out += ch;
  }
  return out;
}

export function formatDate(iso: string): string {

};

export function fitLine(key: FieldKey, text: string): Line {
  const f = FIT[key];
  let size = f.basePt;
  while (size > f.minPt && textWidthPt(text, f.font, size) > f.innerPt) size -= 0.5;
  const overflow = textWidthPt(text, f.font, size) > f.innerPt;
  return { text, sizePt: size, halfPts: Math.round(size * 2), overflow };
}

export function computeLines(data: RegularIdData): Record<FieldKey, Line> {
  const out = {} as Record<FieldKey, Line>;
  (Object.keys(FIT) as FieldKey[]).forEach((k) => {
    out[k] = fitLine(k, displayText(k, data));
  });
  return out;
}

/* -------------------------------------------------------------------------- */
/*  Filling the Word template                                                 */
/*                                                                            */
/*  The card is a DrawingML group (with a VML fallback copy). Its 16 text     */
/*  boxes appear in the same order in both copies, so we patch box N in each. */
/* -------------------------------------------------------------------------- */

const TEXTBOXES_PER_COPY = 16;

/** Index of each text box inside the template. */
const BOX = {
  addressLine2: 0,
  values: 2,
  emergency: 3,
  name: 12,
  designation: 13,
  baseBranch: 14,
  employeeNo: 15,
} as const;

const VALUE_ORDER: FieldKey[] = [
  "dateHired",
  "sss",
  "tin",
  "hdmf",
  "phic",
  "birthdate",
  "addressLine1",
];

const esc = (s: string) =>
  s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");

function setSize(rPr: string, halfPts: number): string {
  const size = `<w:sz w:val="${halfPts}"/><w:szCs w:val="${halfPts}"/>`;
  if (!rPr) return `<w:rPr>${size}</w:rPr>`;
  const stripped = rPr
    .replace(/<w:sz\s[^>]*\/>/g, "")
    .replace(/<w:szCs\s[^>]*\/>/g, "");
  return stripped.replace("</w:rPr>", `${size}</w:rPr>`);
}

/** Replace a paragraph's text, keeping its paragraph and first-run formatting. */
function rewriteParagraph(p: string, line: Line, baseHalfPts: number): string {
  const open = /^<w:p(?:\s[^>]*)?>/.exec(p)?.[0];
  if (!open || !p.endsWith("</w:p>")) return p;
  const pPr = /<w:pPr>[\s\S]*?<\/w:pPr>/.exec(p)?.[0] ?? "";
  const rest = p.slice(open.length, p.length - "</w:p>".length).replace(pPr, "");
  let rPr = /<w:r(?:\s[^>]*)?>\s*(<w:rPr>[\s\S]*?<\/w:rPr>)/.exec(rest)?.[1] ?? "";
  if (!rPr) rPr = /<w:rPr>[\s\S]*?<\/w:rPr>/.exec(pPr)?.[0] ?? "";
  // Only touch the size when the text had to shrink, so untouched boxes stay byte-identical.
  if (line.halfPts !== baseHalfPts) rPr = setSize(rPr, line.halfPts);
  return `${open}${pPr}<w:r>${rPr}<w:t xml:space="preserve">${esc(line.text)}</w:t></w:r></w:p>`;
}

const PARAGRAPH = /<w:p(?:\s[^>]*)?>[\s\S]*?<\/w:p>|<w:p(?:\s[^>]*)?\/>/g;

function rewriteBox(inner: string, keys: FieldKey[], lines: Record<FieldKey, Line>): string {
  let i = 0;
  return inner.replace(PARAGRAPH, (p) => {
    const key = keys[i++];
    if (!key) return p;
    return rewriteParagraph(p, lines[key], Math.round(FIT[key].basePt * 2));
  });
}

function patchDocumentXml(xml: string, data: RegularIdData): string {
  const lines = computeLines(data);
  const plan: Record<number, FieldKey[]> = {
    [BOX.addressLine2]: ["addressLine2"],
    [BOX.values]: VALUE_ORDER,
    [BOX.emergency]: ["emergencyName", "emergencyPhone"],
    [BOX.name]: ["name"],
    [BOX.designation]: ["designation"],
    [BOX.baseBranch]: ["baseBranch"],
    [BOX.employeeNo]: ["employeeNo"],
  };

  let count = 0;
  const patched = xml.replace(
    /<w:txbxContent>([\s\S]*?)<\/w:txbxContent>/g,
    (whole, inner: string) => {
      const idx = count++ % TEXTBOXES_PER_COPY;
      const keys = plan[idx];
      if (!keys) return whole;
      return `<w:txbxContent>${rewriteBox(inner, keys, lines)}</w:txbxContent>`;
    },
  );
  if (count === 0 || count % TEXTBOXES_PER_COPY !== 0) {
    throw new Error(
      `Template layout changed: found ${count} text boxes, expected a multiple of ${TEXTBOXES_PER_COPY}.`,
    );
  }
  return patched;
}

/** The photo is the one picture that sits outside the card group. */
function findPhotoRelId(xml: string): string {
  const withoutGroup = xml.replace(/<mc:AlternateContent>[\s\S]*?<\/mc:AlternateContent>/g, "");
  const m = /<a:blip\s+r:embed="(rId\d+)"/.exec(withoutGroup);
  if (!m) throw new Error("Template layout changed: photo picture not found.");
  return m[1];
}

function replacePhoto(xml: string, relId: string): string {
  // The template crops the sample photo; the uploaded one is already cropped to the frame.
  const re = new RegExp(`(<a:blip\\s+r:embed="${relId}"[\\s\\S]*?</a:blip>)<a:srcRect[^>]*/>`);
  return xml.replace(re, "$1<a:srcRect/>");
}

async function photoTarget(zip: JSZip, relId: string): Promise<string> {
  const rels = await zip.file("word/_rels/document.xml.rels")!.async("string");
  const m = new RegExp(`<Relationship\\s[^>]*Id="${relId}"[^>]*Target="([^"]+)"`).exec(rels)
    ?? new RegExp(`<Relationship\\s[^>]*Target="([^"]+)"[^>]*Id="${relId}"`).exec(rels);
  if (!m) throw new Error("Template layout changed: photo relationship not found.");
  return `word/${m[1]}`;
}

export async function fillRegularTemplate(
  template: ArrayBuffer | Uint8Array,
  data: RegularIdData,
  photoPng: Uint8Array | null,
): Promise<Uint8Array> {
  const zip = await JSZip.loadAsync(template);
  const docFile = zip.file("word/document.xml");
  if (!docFile) throw new Error("Not a valid Word template.");

  let xml = patchDocumentXml(await docFile.async("string"), data);

  if (photoPng) {
    const relId = findPhotoRelId(xml);
    const target = await photoTarget(zip, relId);
    xml = replacePhoto(xml, relId);
    zip.file(target, photoPng);
  }

  zip.file("word/document.xml", xml);
  return zip.generateAsync({ type: "uint8array", compression: "DEFLATE" });
}

export function fileNameFor(data: RegularIdData): string {
  const base = cleanText(data.name).replace(/[\\/:*?"<>|]+/g, "").trim() || "EMPLOYEE";
  return `ID - ${base}.docx`;
}

######### File: README.md #########
# ID Generator — Regular Employee

Type the employee's details, check the live preview, download the ID as a Word file.
Everything runs in the browser: no server processing, nothing uploaded.

## Run it

```bash
npm install
npm run dev        # http://localhost:3000
npm run build && npm start
```

## How it matches your Word file exactly

The app does **not** redraw the card. `public/templates/regular-id.docx` is your original
`REGULAR ID.docx` with the personal values blanked and the sample photo replaced by a neutral
placeholder. On download, only the text boxes (and the photo) are filled in, so the layout,
fonts, crop marks and artwork are byte-for-byte your design. Filling your original file with its own
values re-renders pixel-identical to the original.

Text that is too long for its box is shrunk automatically (down to a minimum size per field,
using Calibri/Arial metrics); the form warns when something still won't fit.

## Project layout

| Path | What it is |
| --- | --- |
| `lib/regularId.ts` | Data model, text fitting, and the `.docx` filler (JSZip) |
| `lib/cardGeometry.ts` | Positions of the boxes, used by the preview |
| `components/` | Form, live preview, photo crop/zoom |
| `public/templates/regular-id.docx` | Sanitised Word template |
| `public/card/regular-blank.png` | Blank card render used behind the preview |
| `scripts/make-template.mjs` | Rebuilds the template from the original Word file |

## If you change the Word design

```bash
node scripts/make-template.mjs "REGULAR ID.docx" --transparent-photo /tmp/bg.docx
```

then re-render `/tmp/bg.docx` at 300 dpi (LibreOffice → PDF → PNG), crop to
`public/card/regular-blank.png`, and re-check `lib/cardGeometry.ts` if boxes moved.
The filler expects the same 16 text boxes in the same order; it throws a clear error if not.

## Adding the other ID types

Copy `lib/regularId.ts` + `components/RegularIdGenerator.tsx` per type, point them at that type's
template, and update the box indexes in `BOX`.
